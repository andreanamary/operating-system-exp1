#include <stdio.h>

int main() {
    int n, m;
    printf("Enter number of pages in reference string: \n");
    scanf("%d", &n);

    int ref[n];
    printf("Enter the reference string: \n");
    for (int i = 0; i < n; i++) {
        scanf("%d", &ref[i]);
    }

    printf("Enter number of frames: \n");
    scanf("%d", &m);

    int frames[m];
    for (int i = 0; i < m; i++) frames[i] = -1;

    int faults = 0, hits = 0;
    int index = 0; // FIFO pointer

    for (int i = 0; i < n; i++) {
        int page = ref[i];
        int found = 0;

        // Check if page is already in frames
        for (int j = 0; j < m; j++) {
            if (frames[j] == page) {
                found = 1;
                break;
            }
        }

        if (found) {
            hits++;
        } else {
            frames[index] = page;
            index = (index + 1) % m;
            faults++;
        }

    }

    printf("\nTotal Page Faults: %d\n", faults);
    printf("Total Page Hits: %d\n", hits);

    return 0;
}
