#include <stdio.h>

int main() {
    int n, i;
    float totTAT = 0, totWT = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int AT[n], BT[n], CT[n], TAT[n], WT[n];

    for(int i = 0; i < n; i++) {
        printf("\nEnter Arrival Time and Burst Time for P%d: ", i + 1);
        scanf("%d %d", &AT[i], &BT[i]);
    }

    CT[0] = AT[0] + BT[0];

    for(int i = 1; i < n; i++) {
        if(CT[i - 1] < AT[i])
            CT[i] = AT[i] + BT[i];
        else
            CT[i] = CT[i - 1] + BT[i];
    }

    for(int i = 0; i < n; i++) {
        TAT[i] = CT[i] - AT[i];
        WT[i] = TAT[i] - BT[i];
        totTAT += TAT[i];
        totWT += WT[i];
    }

    printf("\n\nAverage Turnaround Time: %.2f", totTAT / n);
    printf("\nAverage Waiting Time: %.2f\n", totWT / n);

    return 0;
}
